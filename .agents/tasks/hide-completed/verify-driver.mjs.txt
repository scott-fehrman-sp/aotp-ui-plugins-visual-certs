function check(name, cond, detail = '') {

/* ---------------- page-side helpers ---------------- */
function installHelpers() {
  const q = (s, root = document) => [...root.querySelectorAll(s)]
  const rectOf = (el) => {
    const r = el.getBoundingClientRect()
    return { l: r.left, t: r.top, r: r.right, b: r.bottom, w: r.width, h: r.height }
  }
  const edgeMap = new Map()
  for (const e of window.__graph.edges) edgeMap.set(`e:${e.from}:${e.to}`, e)
  window.__vpLog = []
  const vp = document.querySelector('.react-flow__viewport')
  window.__hc = {
    zoom() {
      const m = /scale\(([-\d.]+)\)/.exec(document.querySelector('.react-flow__viewport').style.transform)
      return m ? +m[1] : null
    },
    settle() {
      return new Promise((res) => {
        let last = null
        let same = 0
        let frames = 0
        const tick = () => {
          frames++
          const t = document.querySelector('.react-flow__viewport')?.style.transform
          if (t === last) same++
          else {
            same = 0
            last = t
          }
          if (frames > 4 && same >= 8) res(t)
          else requestAnimationFrame(tick)
        }
        tick()
      })
    },
    startVpLog() {
      window.__vpLog = []
      let last = document.querySelector('.react-flow__viewport').style.transform
      window.__vpObs?.disconnect()
      window.__vpObs = new MutationObserver(() => {
        const t = document.querySelector('.react-flow__viewport').style.transform
        if (t !== last) {
          window.__vpLog.push(t)
          last = t
        }
      })
      window.__vpObs.observe(document.querySelector('.react-flow__viewport'), { attributes: true })
    },
    vpChanges() {
      return window.__vpLog.length
    },
    collect() {
      const canvas = rectOf(document.querySelector('.react-flow'))
      const nodes = q('.react-flow__node').map((n) => {
        const card = n.querySelector('.vc-graph-card')
        const m = /translate\((-?[\d.]+)px,\s*(-?[\d.]+)px\)/.exec(n.style.transform)
        const cs = getComputedStyle(card)
        return {
          id: n.dataset.id,
          label: card.querySelector('.vc-graph-card-label').textContent,
          kind: [...card.classList].find((c) => c.startsWith('kind-')).slice(5),
          status: card.dataset.reviewStatus || null,
          reference: card.classList.contains('vc-graph-card-reference'),
          pressed: card.getAttribute('aria-pressed'),
          hasIcon: !!card.querySelector('.vc-graph-status-icon'),
          gx: +m[1],
          gy: +m[2],
          rect: rectOf(n),
          borderRight: cs.borderLeftColor,
          borderRightW: cs.borderLeftWidth,
          outline: (/^(rgba?\([^)]*\))/.exec(cs.boxShadow) || [])[1] || null,
          boxShadow: cs.boxShadow,
          opacity: cs.opacity,
        }
      })
      const edges = q('.react-flow__edge').map((e) => {
        const id = (e.dataset.id || '').replace(/^rf__edge-/, '')
        const path = e.querySelector('.react-flow__edge-path')
        const info = edgeMap.get(id)
        return {
          id,
          from: info?.from,
          to: info?.to,
          cls: e.getAttribute('class'),
          stroke: path ? getComputedStyle(path).stroke : null,
          inlineOpacity: path ? path.style.opacity : null,
        }
      })
      const overlays = [
        ...q('.react-flow__panel').map((p) => ({ name: 'panel:' + [...p.classList].filter((c) => c !== 'react-flow__panel').join('.'), ignore: p.classList.contains('vc-graph-fit-ignore'), ...rectOf(p), text: p.textContent.trim().slice(0, 60) })),
        ...q('.react-flow__minimap').map((p) => ({ name: 'minimap', ignore: false, ...rectOf(p) })),
        ...q('.react-flow__controls').map((p) => ({ name: 'controls', ignore: false, ...rectOf(p) })),
      ]
      const wrap = rectOf(document.querySelector('.vc-graph-wrap'))
      return {
        canvas,
        wrap,
        nodes,
        edges,
        overlays,
        zoom: window.__hc.zoom(),
        scrollH: document.documentElement.scrollHeight,
        innerH: window.innerHeight,
        innerW: window.innerWidth,
        scrollW: document.documentElement.scrollWidth,
        hint: document.querySelector('.vc-graph-hint[role="status"]')?.textContent ?? null,
        notes: q('.vc-graph-note').map((n) => ({ text: n.textContent, panel: n.closest('.react-flow__panel')?.className, rect: rectOf(n) })),
        live: q('[aria-live="polite"]').map((n) => n.textContent),
        legend: document.querySelector('.vc-graph-legend')?.textContent ?? null,
      }
    },
    toolbar() {
      return q('[role="toolbar"] button').map((b) => ({
        label: b.getAttribute('aria-label'),
        pressed: b.getAttribute('aria-pressed'),
        disabled: b.disabled,
        active: b.classList.contains('vc-graph-control-active'),
        icon: b.querySelector('svg')?.dataset.icon,
        w: b.getBoundingClientRect().width,
        h: b.getBoundingClientRect().height,
        radius: getComputedStyle(b).borderRadius,
        border: getComputedStyle(b).border,
        bg: getComputedStyle(b).backgroundColor,
        color: getComputedStyle(b).color,
        opacity: getComputedStyle(b).opacity,
      }))
    },
  }
}

async function openPage(fixture, w, h, opts = {}) {
  const page = await browser.newPage()
  await page.setViewport({ width: w, height: h, deviceScaleFactor: 1 })
  if (opts.dark) await page.emulateMediaFeatures([{ name: 'prefers-color-scheme', value: 'dark' }])
  page.on('pageerror', (e) => check('no page error', false, String(e)))
  page.on('console', (m) => {
    if (m.type() === 'error' && !/favicon|Download the React DevTools|404/.test(m.text())) check('no console error', false, m.text())
  })
  await page.goto(`${BASE}?fixture=${fixture}`, { waitUntil: 'networkidle0' })
  await page.waitForSelector('.vc-graph-wrap .react-flow__node')
  await page.evaluate(installHelpers)
  await page.mouse.move(2, 2)
  return page
}

async function ensure(page, kind, want) {
  // kind: 'reference' | 'completed' | 'layout'
  const tb = await T(page)
  if (kind === 'layout') {
    const isLR = tb[0].label === 'Layout: vertical'

    if ((b.pressed === 'true') !== want) await clickLabel(page, b.label)
  }
}

/* ---------------- structural checks ---------------- */
function reachable(userId, edges) {
  const out = new Map()
  for (const e of edges) (out.get(e.from) ?? out.set(e.from, []).get(e.from)).push(e)
  const seen = new Set([userId])
  const q = [userId]
  while (q.length) for (const e of out.get(q.shift()) ?? []) if (!seen.has(e.to)) { seen.add(e.to); q.push(e.to) }
  return seen
}

function checkHide(s0, s1, userId, tag) {
  const byId0 = new Map(s0.nodes.map((n) => [n.id, n]))
  const userEdges = s0.edges.filter((e) => e.from === userId)
  const completedTargets = new Set(userEdges.filter((e) => COMPLETED.has(byId0.get(e.to)?.status)).map((e) => e.to))
  const remaining = s0.edges.filter((e) => !(e.from === userId && completedTargets.has(e.to)))
  const reach = reachable(userId, remaining)
  const expectedEdges = remaining.filter((e) => reach.has(e.from)).map((e) => e.id).sort()
  const got = s1.nodes.map((n) => n.id).sort()
  const exp = [...reach].sort()
  check(`C2 ${tag} node set == oracle`, JSON.stringify(got) === JSON.stringify(exp), `got=${got.length} exp=${exp.length} diff=${[...got.filter((x) => !exp.includes(x)), ...exp.filter((x) => !got.includes(x))].join(',')}`)
  const gotE = s1.edges.map((e) => e.id).sort()
  check(`C2 ${tag} edge set == oracle`, JSON.stringify(gotE) === JSON.stringify(expectedEdges), `got=${gotE.length} exp=${expectedEdges.length}`)
  // C1
  const byId1 = new Map(s1.nodes.map((n) => [n.id, n]))
  const completedFirst = s1.nodes.filter((n) => !n.reference && COMPLETED.has(n.status))
  check(`C1 ${tag} no completed reviewable card remains`, completedFirst.length === 0, completedFirst.map((n) => n.label).join(','))
  const needs0 = s0.nodes.filter((n) => !n.reference && n.status === 'needsReview')
  check(`C1 ${tag} every needs-review card remains`, needs0.every((n) => byId1.has(n.id)), needs0.filter((n) => !byId1.has(n.id)).map((n) => n.label).join(','))
  // demoted nodes
  for (const id of completedTargets) {
    const n = byId1.get(id)
    if (n) {
      check(`C3 ${tag} demoted '${n.label}' is muted reference without icon`, n.reference && !n.hasIcon && !n.status, JSON.stringify({ r: n.reference, i: n.hasIcon, s: n.status }))
      const inc = s1.edges.filter((e) => e.to === id)
      check(`C3 ${tag} demoted '${n.label}' has no edge from the user`, inc.every((e) => e.from !== userId))
    }
  }
  // C3 consistency of colors
  const colorByStatus = {}
  for (const n of s1.nodes.filter((x) => !x.reference && x.status)) {
    const edge = s1.edges.find((e) => e.from === userId && e.to === n.id)
    check(`C3 ${tag} '${n.label}' has its status edge`, !!edge)
    if (!edge) continue
    check(`C3 ${tag} '${n.label}' edge class matches status`, edge.cls.includes(`vc-graph-edge-${n.status}`), edge.cls)
    const sig = [n.borderRight, n.outline, edge.stroke].join('|')
    check(`C3 ${tag} '${n.label}' border/outline/edge colors agree`, n.borderRight === edge.stroke && n.outline === edge.stroke, sig)
    colorByStatus[n.status] ??= edge.stroke
    check(`C3 ${tag} '${n.label}' same color as other ${n.status} items`, colorByStatus[n.status] === edge.stroke)
  }
  const distinct = new Set(Object.values(colorByStatus))
  check(`C3 ${tag} distinct colors per status`, distinct.size === Object.keys(colorByStatus).length)
  check(`C3 ${tag} no completed status edge remains`, s1.edges.every((e) => !(e.from === userId && completedTargets.has(e.to))))
  check(`C3 ${tag} no edge with missing endpoint`, s1.edges.every((e) => byId1.has(e.from) && byId1.has(e.to)))
}

const KIND_ORDER = ['user', 'role', 'accessProfile', 'entitlement', 'source']
function checkLanes(s, rankdir, tag) {
  const stride = rankdir === 'LR' ? 112 : 248
  const rankKey = rankdir === 'LR' ? 'gx' : 'gy'
  const crossKey = rankdir === 'LR' ? 'gy' : 'gx'
  const kinds = KIND_ORDER.filter((k) => s.nodes.some((n) => n.kind === k))
  let prev = -Infinity
  for (const k of kinds) {
    const ns = s.nodes.filter((n) => n.kind === k)
    const ranks = new Set(ns.map((n) => n[rankKey]))
    check(`C4 ${tag} ${k} in one lane`, ranks.size === 1, [...ranks].join(','))
    const r = ns[0][rankKey]
    check(`C4 ${tag} lane order ${k}`, r > prev, `${r} <= ${prev}`)
    prev = r
    const cross = ns.map((n) => n[crossKey]).sort((a, b) => a - b)
    for (let i = 1; i < cross.length; i++) check(`C4 ${tag} ${k} evenly spaced`, Math.abs(cross[i] - cross[i - 1] - stride) < 0.01, `${cross[i] - cross[i - 1]}`)
  }
  for (let i = 0; i < s.nodes.length; i++)
    for (let j = i + 1; j < s.nodes.length; j++)
      if (intersects(s.nodes[i].rect, s.nodes[j].rect, 0.5)) check(`C4 ${tag} no card overlap`, false, `${s.nodes[i].label}/${s.nodes[j].label}`)
}

function checkFit(s, tag, { strictAllowed = true } = {}) {
  const readable = s.zoom !== null && Math.abs(s.zoom - 0.65) < 1e-6
  const clamp = s.zoom !== null && Math.abs(s.zoom - 0.15) < 1e-6
  const ov = s.overlays.filter((o) => !o.ignore)
  const user = s.nodes.find((n) => n.kind === 'user')
  if (user) for (const o of ov) check(`C5 ${tag} user card clear of ${o.name}`, !intersects(user.rect, o, 0.5), o.name)
  if (readable || clamp) {
    if (clamp) report(`${tag}: zoom clamped at minZoom 0.15 (existing clamp)`)
    if (readable) report(`${tag}: readable-zoom (scroll) mode, strict containment not applicable`)
    return
  }
  if (!strictAllowed) return
  for (const n of s.nodes) {
    check(`C5 ${tag} '${n.label}' inside canvas`, n.rect.l >= s.canvas.l - 1 && n.rect.r <= s.canvas.r + 1 && n.rect.t >= s.canvas.t - 1 && n.rect.b <= s.canvas.b + 1, JSON.stringify(n.rect))
    for (const o of ov) if (intersects(n.rect, o, 0.5)) check(`C5 ${tag} '${n.label}' not under ${o.name}`, false, `${o.name} ${o.text}`)
  }
}

function checkPage(s, tag) {
  check(`C9 ${tag} no page scroll`, s.scrollH <= s.innerH + 1 && s.scrollW <= s.innerW + 1, `scrollH=${s.scrollH} innerH=${s.innerH}`)
  check(`C9 ${tag} canvas ends at window bottom`, s.wrap.b <= s.innerH + 1 && s.wrap.b >= s.innerH - 30, `bottom=${s.wrap.b} inner=${s.innerH}`)
}

    }

}

async function runSelection() {
  ctx = 'selection mixed 1700x1000'
  const page = await openPage('mixed', 1700, 1000)
  const sid = await page.evaluate(() => window.__graph.nodes.find((n) => n.label === 'User One').id)
  await drillIn(page, 'User One')
  const clickCard = async (label) => {
    const h = await page.evaluateHandle((l) => [...document.querySelectorAll('.vc-graph-card')].find((c) => c.querySelector('.vc-graph-card-label').textContent === l), label)
    await h.asElement().click()
    await page.mouse.move(2, 2)
    await sleep(60)
  }
  // completed item selected, then hidden
  await clickCard('AP Billing')
  let s = await S(page)
  check('C7 select completed: pressed', s.nodes.filter((n) => n.pressed === 'true').map((n) => n.label).join() === 'AP Billing')

  check('C7 hiding the selected item clears the selection', !s.nodes.some((n) => n.pressed === 'true'))
  check('C7 no highlighted/lineage edges', !s.edges.some((e) => /vc-graph-edge-(selected|lineage)/.test(e.cls)))
  check('C7 no dimmed edges', !s.edges.some((e) => e.inlineOpacity === '0.25' || e.inlineOpacity === '0.12'))
  check('C7 no stale Highlighting announcement', !/Highlighting/.test(s.live[1]), s.live[1])
  await clickLabel(page, 'Show completed items', { prefix: true })
  s = await S(page)
  check('C7 show again does not re-select', !s.nodes.some((n) => n.pressed === 'true') && !s.edges.some((e) => /vc-graph-edge-selected/.test(e.cls)))
  // visible item survives
  await clickCard('Role B')
  s = await S(page)
  const before = s.edges.filter((e) => /vc-graph-edge-(selected|lineage)/.test(e.cls)).map((e) => e.id).sort()
  check('C7 Role B selected with highlight', s.nodes.filter((n) => n.pressed === 'true').map((n) => n.label).join() === 'Role B' && before.length >= 2, before.join())
  await clickLabel(page, 'Hide completed items')
  s = await S(page)
  const after = s.edges.filter((e) => /vc-graph-edge-(selected|lineage)/.test(e.cls))
  check('C7 visible selection survives hide', s.nodes.filter((n) => n.pressed === 'true').map((n) => n.label).join() === 'Role B')
  const roleB = s.nodes.find((n) => n.label === 'Role B')
  check('C8 highlight colors == status color', after.length >= 2 && after.every((e) => e.stroke === roleB.borderRight), JSON.stringify([after.map((e) => e.stroke), roleB.borderRight]))
  // every remaining edge below Role B (its lineage) highlighted: the edges whose source is reachable from Role B
  const reach = reachable(roleB.id, s.edges)
  const expectLineage = s.edges.filter((e) => reach.has(e.from) && e.from !== sid).map((e) => e.id).concat(s.edges.filter((e) => e.from === sid && e.to === roleB.id).map((e) => e.id)).sort()
  check('C7 whole visible lineage highlighted', JSON.stringify(after.map((e) => e.id).sort()) === JSON.stringify(expectLineage), JSON.stringify([after.map((e) => e.id).sort(), expectLineage]))

  s = await S(page)
  check('C7 visible selection survives show', s.nodes.filter((n) => n.pressed === 'true').map((n) => n.label).join() === 'Role B')

async function runLong() {
  // C10 scroll hint + readable zoom
  for (const [w, h] of VPS) {
    ctx = `long ${w}x${h}`
    const page = await openPage('long', w, h)
    await drillIn(page, 'Dense User')
    const sShown = await S(page)
    await ensure(page, 'reference', false)
    const sRef = await S(page)
    await ensure(page, 'reference', true)
    report(`long shown: zoom=${sShown.zoom} hint=${JSON.stringify(sShown.hint)} first-level cards in DOM=${firstLevelCount(sShown)}`)
    check('C10 long shown: readable zoom + hint', sShown.hint !== null && Math.abs(sShown.zoom - 0.65) < 0.005, `${sShown.hint} ${sShown.zoom}`)
    await clickLabel(page, 'Hide completed items')
    const s1 = await S(page)
    const total = await page.evaluate(() => window.__graph.edges.filter((e) => e.firstLevel).length)
    report(`long hidden: zoom=${s1.zoom} hint=${JSON.stringify(s1.hint)} first-level cards in DOM=${firstLevelCount(s1)}`)
    check('C10 long hidden: still readable zoom with a smaller count', s1.hint !== null && Math.abs(s1.zoom - 0.65) < 0.005, `${s1.hint} ${s1.zoom}`)
    const m = /^(\d+) items/.exec(s1.hint ?? '')
    const m0 = /^(\d+) items/.exec(sShown.hint ?? '')
    check('C10 hint counts visible first-level items (smaller than shown)', m && m0 && +m[1] < +m0[1], `${s1.hint} < ${sShown.hint}`)
    // expected visible: needsReview first-level items of u1 (computed from graph)
    const expectedVisible = await page.evaluate(() => {
      const g = window.__graph
      const u = g.nodes.find((n) => n.label === 'Dense User').id
      return g.edges.filter((e) => e.from === u && e.firstLevel && e.reviewStatus === 'needsReview').length
    })
    check('C10 hint count == visible first-level items per graph data', m && +m[1] === expectedVisible, `${s1.hint} vs ${expectedVisible}`)
    await ensure(page, 'reference', false)
    const s2 = await S(page)
    const m2 = /^(\d+) items/.exec(s2.hint ?? '')
    report(`long hidden+ref hidden: hint=${JSON.stringify(s2.hint)} zoom=${s2.zoom}`)
    check('C10 hint count with reference hidden too', !m2 || +m2[1] === expectedVisible, s2.hint)
    await page.close()
  }
  // mostlyDone: shrinks to <10 -> hint disappears
  ctx = 'mostlyDone 1700x1000'
  const page = await openPage('mostlyDone', 1700, 1000)
  await drillIn(page, 'Dense User')
  const s0 = await S(page)
  await clickLabel(page, 'Hide completed items')
  const s1 = await S(page)
  report(`mostlyDone shown hint=${JSON.stringify(s0.hint)} zoom=${s0.zoom}; hidden hint=${JSON.stringify(s1.hint)} zoom=${s1.zoom} cards=${s1.nodes.length}`)
  check('C10 hint disappears once few items remain', s1.hint === null)
  checkFit(s1, 'mostlyDone hidden')
  await page.close()
}
