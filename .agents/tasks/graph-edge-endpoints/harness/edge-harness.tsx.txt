// Record, for every card wrapper, the class list and the card's computed
// transform AT THE MOMENT IT FIRST APPEARED in the DOM.
const first: Record<string, { cls: string; cardTransform: string; t: number }> = {}
;(window as unknown as { __first: typeof first }).__first = first
new MutationObserver((records) => {
  for (const r of records) {
    r.addedNodes.forEach((added) => {
      if (!(added instanceof HTMLElement)) return
      const els = added.matches('.react-flow__node') ? [added] : Array.from(added.querySelectorAll('.react-flow__node'))
      for (const el of els) {
        const id = el.getAttribute('data-id')!
        if (!id || first[id]) continue
        const card = el.querySelector('.vc-graph-card')
        first[id] = { cls: el.className, cardTransform: card ? getComputedStyle(card).transform : 'none', t: performance.now() }
      }
    })
  }
}).observe(document, { childList: true, subtree: true })
